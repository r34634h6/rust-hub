'use client';
import { useState } from 'react';

const structures = [
  { id: 'wood-door', name: 'Holztür', cheapest: '18 Explo Ammo oder Propane Bomb' },
  { id: 'stone-wall', name: 'Steinwand', cheapest: '1 C4 + 82 Explo' },
  { id: 'metal-wall', name: 'Metallwand', cheapest: '3 C4' },
  { id: 'armored', name: 'Panzerplatte', cheapest: '2-3 C4' },
];

export default function RaidCalculator() {
  const [selected, setSelected] = useState(structures[0]);
  return (
    <div className="bg-zinc-900 p-8 rounded-2xl">
      <h2 className="text-3xl font-bold text-orange-400 mb-6">Raid Calculator</h2>
      <div className="grid grid-cols-2 gap-4">
        {structures.map(s => (
          <button key={s.id} onClick={() => setSelected(s)}
            className="p-6 bg-zinc-800 hover:bg-orange-600 rounded-xl text-left">
            {s.name}<br/>
            <span className="text-orange-300 text-sm">{s.cheapest}</span>
          </button>
        ))}
      </div>
    </div>
  );
}
